     group=grp_panel)

tbl_size_input = input.string("Normal", "Yazı & Panel Boyutu", 
     options=["Çok Küçük (Tiny)", "Küçük (Small)", "Normal", "Büyük (Large)", "Ekstra Büyük (Huge)"], 
     tooltip="Yazıların ve panelin boyutunu buradan büyütebilirsiniz.", group=grp_panel)

bar_len_input  = input.int(10, "İlerleme Çubuğu Uzunluğu (Bar Boyutu)", minval=6, maxval=24, group=grp_panel)
show_comment   = input.bool(true, "Otomatik Piyasa Yorumunu Göster", group=grp_panel)

grp_oi    = "📈 Open Interest & Fonlama Ayarları"
oi_len    = input.int(14, "OI Trend Periyodu (SMA)", minval=1, group=grp_oi)
custom_oi = input.string("", "Özel OI Ticker (Opsiyonel)", group=grp_oi)

// ==========================================
// 2. YARDIMCI FONKSİYONLAR
// ==========================================
// Sayı Kısaltma (1.46B, 17.33M, 45K)
f_format_num(float val) =>
    if na(val)
        "N/A"
    else if math.abs(val) >= 1000000000
        str.tostring(val / 1000000000, "#.##") + "B"
    else if math.abs(val) >= 1000000
        str.tostring(val / 1000000, "#.##") + "M"
    else if math.abs(val) >= 1000
        str.tostring(val / 1000, "#.##") + "K"
    else
        str.tostring(val, "#.##")

// Kompakt Kesintisiz Bar Üretici
f_make_bars(float l_pct, int total_len) =>
    int l_cnt = math.min(total_len - 1, math.max(1, math.round((l_pct / 100.0) * total_len)))
    int s_cnt = total_len - l_cnt
    string bar_char = "━"
    string l_str = ""
    string s_str = ""
    if l_cnt > 0
        for i = 1 to l_cnt
            l_str := l_str + bar_char
    if s_cnt > 0
        for j = 1 to s_cnt
            s_str := s_str + bar_char
    [l_str, s_str]

// Konum Seçici
f_get_pos(string p) =>
    switch p
        "Sol Alt (Bottom Left)"    => position.bottom_left
        "Sol Üst (Top Left)"       => position.top_left
        "Sağ Üst (Top Right)"      => position.top_right
        "Sağ Alt (Bottom Right)"   => position.bottom_right
        "Sol Orta (Middle Left)"   => position.middle_left
        "Sağ Orta (Middle Right)"  => position.middle_right
        "Üst Orta (Top Center)"    => position.top_center
        "Alt Orta (Bottom Center)" => position.bottom_center
        => position.bottom_right

// Font Boyutu Seçici
f_get_size(string s) =>
    switch s
        "Çok Küçük (Tiny)"     => size.tiny
        "Küçük (Small)"        => size.small
        "Normal"               => size.normal
        "Büyük (Large)"        => size.large
        "Ekstra Büyük (Huge)"  => size.huge
        => size.normal

// ==========================================
// 3. VERİLERİ ÇEKME (TRADER POS, EXPOSURE, OI, FR)
// ==========================================
clean_prefix = syminfo.prefix != "" ? syminfo.prefix : "BINANCE"
clean_base   = syminfo.basecurrency != "" ? syminfo.basecurrency : "BTC"

// 3.1 Trader Positioning (Account L/S Ratio)
sym_acc_ls = "BINANCE:" + clean_base + "USDT_ACCOUNT_LS_RATIO"
acc_ls_val = request.security(sym_acc_ls, timeframe.period, close, ignore_invalid_symbol=true)

// 3.2 Market Exposure (Positions L/S Ratio)
sym_pos_ls = "BINANCE:" + clean_base + "USDT_POSITIONS_LS_RATIO"
pos_ls_val = request.security(sym_pos_ls, timeframe.period, close, ignore_invalid_symbol=true)

// Fallback Dinamiği
candle_range  = high - low
bull_part     = candle_range == 0 ? 0.5 : (close - low) / candle_range
bear_part     = candle_range == 0 ? 0.5 : (high - close) / candle_range
calc_buy_vol  = ta.ema(volume * bull_part, 5)
calc_sell_vol = ta.ema(volume * bear_part, 5)
calc_total    = calc_buy_vol + calc_sell_vol
fallback_long = calc_total > 0 ? (calc_buy_vol / calc_total) * 100.0 : 50.0

float tp_long_pct = 50.0
float tp_short_pct = 50.0
if not na(acc_ls_val) and acc_ls_val > 0
    tp_long_pct  := (acc_ls_val / (acc_ls_val + 1.0)) * 100.0
    tp_short_pct := 100.0 - tp_long_pct
else
    tp_long_pct  := fallback_long
    tp_short_pct := 100.0 - tp_long_pct

float me_long_pct = 50.0
float me_short_pct = 50.0
if not na(pos_ls_val) and pos_ls_val > 0
    me_long_pct  := (pos_ls_val / (pos_ls_val + 1.0)) * 100.0
    me_short_pct := 100.0 - me_long_pct
else
    me_long_pct  := fallback_long
    me_short_pct := 100.0 - me_long_pct

// 3.3 Open Interest (OI)
oi_clean_1 = clean_prefix + ":" + clean_base + "USDT.P_OI"
oi_clean_2 = "BINANCE:" + clean_base + "USDT.P_OI"
oi_clean_3 = "BYBIT:" + clean_base + "USDT.P_OI"

oi_target  = custom_oi != "" ? custom_oi : oi_clean_1

oi_data = request.security(oi_target, timeframe.period, close, ignore_invalid_symbol=true)
if na(oi_data)
    oi_data := request.security(oi_clean_2, timeframe.period, close, ignore_invalid_symbol=true)
if na(oi_data)
    oi_data := request.security(oi_clean_3, timeframe.period, close, ignore_invalid_symbol=true)

oi_prev   = nz(oi_data[1], oi_data)
oi_diff   = oi_data - oi_prev
oi_pct    = oi_prev > 0 ? (oi_diff / oi_prev) * 100.0 : 0.0
bool is_oi_positive = oi_pct >= 0

// 3.4 Funding Rate (FR - Fonlama Oranı)

sym_fr_2 = "BINANCE:" + clean_base + "USDT_FUNDING_RATE"

if na(fr_data)
    fr_data := request.security(sym_fr_2, timeframe.period, close, ignore_invalid_symbol=true)
if na(fr_data)
    fr_data := request.security(sym_fr_3, timeframe.period, close, ignore_invalid_symbol=true)

// Fonlama Oranı Yüzdesi
float fr_pct = na(fr_data) ? 0.0100 : (math.abs(fr_data) < 0.001 ? fr_data * 100.0 : fr_data)

else if not is_price_up and is_oi_positive

    flow_color := color.rgb(255, 65, 95)
